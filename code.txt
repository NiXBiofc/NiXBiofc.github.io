program Task;

var
  a: array of integer; // Объявляем массив
  i: integer;          // Переменная для циклов
  sumOdd: integer;     // Сумма элементов с НЕЧЕТНЫМИ номерами (1-й, 3-й...)
  sumEven: integer;    // Сумма элементов с ЧЕТНЫМИ номерами (2-й, 4-й...)

begin
  // 1. Создаем массив с готовыми числами
  a := array of integer(4, 10, 5, 2, 8, 1, 7, 3);
  
  // Выводим исходный массив на экран
  writeln('Исходный массив:');
  for i := 0 to 7 do
    write(a[i], ' ');
  writeln;

  // Обнуляем суммы перед подсчетом
  sumOdd := 0;
  sumEven := 0;

  // 2. Считаем суммы элементов
  // В таком массиве 1-й элемент имеет индекс 0, 2-й — индекс 1 и так далее.
  for i := 0 to 7 do
  begin
    if i mod 2 = 0 then
      sumOdd := sumOdd + a[i]    // Если индекс четный (0, 2, 4), то номер элемента НЕЧЕТНЫЙ (1, 3, 5)
    else
      sumEven := sumEven + a[i]; // Если индекс нечетный (1, 3, 5), то номер элемента ЧЕТНЫЙ (2, 4, 6)
  end;

  // 3. Заменяем элементы по условию задачи
  for i := 0 to 7 do
  begin
    if i mod 2 = 0 then
      a[i] := sumEven // Нечетный элемент (индекс 0, 2...) меняем на сумму ЧЕТНЫХ
    else
      a[i] := sumOdd;  // Четный элемент (индекс 1, 3...) меняем на сумму НЕЧЕТНЫХ
  end;

  // 4. Выводим измененный массив на экран
  writeln('Измененный массив:');
  for i := 0 to 7 do
    write(a[i], ' ');
  writeln;
end.
