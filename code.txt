var
  a: array[1..5] of integer;
  i, sum: integer;
  S: boolean;

begin
  writeln('Введите 5 элементов массива:');
  for i := 1 to 5 do
  begin
    write('a[', i, '] = ');
    readln(a[i]);
  end;

  sum := 0;
  for i := 2 to 5 do
  begin
    if a[i] mod 2 <> 0 then
      sum := sum + a[i];
  end;

  S := (a[1] = 3 * sum);

  writeln;
  writeln('Сумма нечётных элементов (кроме первого): ', sum);
  writeln('a[1] = ', a[1]);
  writeln('3 * sum = ', 3 * sum);

  if S then
    writeln('Условие ВЫПОЛНЕНО: первый элемент в 3 раза больше суммы остальных нечётных.')
  else
    writeln('Условие НЕ выполнено.');
  
  readln;
end.
