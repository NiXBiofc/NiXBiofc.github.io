var
  a: array[1..5] of integer;
  i, b, sum: integer;
  c: longint;
begin
  for i := 1 to 5 do
  begin
    writeln('a[', i, ']=');
    readln(a[i]);
  end;

  b := 0;
  c := 1;       { <-- было 0 }
  sum := 0;

  for i := 1 to 5 do   { <-- добавлен цикл }
  begin
    if (a[i] mod 2 <> 0) and (i mod 2 = 0) then  { <-- исправлено условие }
    begin
      b := b + 1;
      c := c * a[i];
      sum := sum + a[i];
    end;
  end;

  writeln('Количество: ', b);
  writeln('Произведение: ', c);
  writeln('Сумма: ', sum);
end.
